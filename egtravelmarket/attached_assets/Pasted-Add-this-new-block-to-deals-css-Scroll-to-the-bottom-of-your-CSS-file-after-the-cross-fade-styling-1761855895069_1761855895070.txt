/* ===== Cinematic Background for Flash Offers ===== */
#flash-offers {
  position: relative;
  overflow: hidden;
  padding: 60px 0;
  background: linear-gradient(180deg, #08111f 0%, #0b1220 100%);
  border-top: 1px solid rgba(255,255,255,0.05);
  border-bottom: 1px solid rgba(255,255,255,0.05);
}

#flash-offers::before,
#flash-offers::after {
  content: "";
  position: absolute;
  inset: 0;
  background: radial-gradient(circle at 20% 30%, rgba(44,125,240,0.25), transparent 60%),
              radial-gradient(circle at 80% 70%, rgba(0,212,255,0.25), transparent 60%);
  opacity: 0.5;
  animation: floatGlow 45s linear infinite;
  z-index: 0;
}

#flash-offers::after {
  opacity: 0.35;
  animation-direction: reverse;
  animation-duration: 60s;
}

@keyframes floatGlow {
  0% { transform: translateY(0) rotate(0deg); }
  50% { transform: translateY(-15px) rotate(3deg); }
  100% { transform: translateY(0) rotate(0deg); }
}

#flash-offers::before {
