⚙️ 2. Updated flash-offer.js

Replace your old JS file with this one 👇

// ==========================
// Load Flash Offers from JSON
// ==========================
(async function loadFlashOffers() {
  try {
    const res = await fetch('flash-offers.json');
    const offers = await res.json();
    const track = document.querySelector('.flash-track');
    if (!track) return;

    track.innerHTML = ''; // clear old placeholder

    offers.forEach(o => {
      const card = document.createElement('article');
      card.className = 'flash-card';
      card.innerHTML = `
        <span class="badge">${o.badge || 'Flash Offer'}</span>
        <h3>${o.title}</h3>
        <p class="desc">${o.desc}</p>
        <div class="countdown" data-deadline="${o.deadline}">Ends in —</div>
        <div class="actions">
          <a href="${o.link}" class="btn">Book Now</a>
          <a href="${o.link}#details" class="link">Details</a>
        </div>
      `;
      track.appendChild(card);
    });

    initCountdowns();
    initSlider(track);
    markFlashBadges(offers);

  } catch (err) {
    console.error('Could not load flash-offers.json', err);
  }
})();

// ==========================
// Countdown
// ==========================
function initCountdowns() {
  const els = Array.from(document.querySelectorAll('.countdown[data-deadline]'));
  function two(n){return n<10?'0'+n:n;}

  function tick(){
    const now = new Date().getTime();
    els.forEach(el=>{
      const end = new Date(el.dataset.deadline).getTime();
      const diff = end - now;
      if(isNaN(end)){el.textContent='—';return;}
      if(diff<=0){el.textContent='Offer ended';return;}
      const d=Math.floor(diff/86400000);
      const h=Math.floor((diff/3600000)%24);
      const m=Math.floor((diff/60000)%60);
      el.textContent=`Ends in ${two(d)}d ${two(h)}h ${two(m)}m`;
    });
  }
  tick();
  setInterval(tick,30000);
}

// ==========================
// Simple Auto Slider
// ==========================
function initSlider(track){
  if(!track || track.children.length<=1) return;
  const interval=6000;
  const cards=Array.from(track.children);
  let index=0;
  function next(){
    index=(index+1)%cards.length;
    cards[index].scrollIntoView({behavior:'smooth',inline:'start'});
  }
  let timer=setInterval(next,interval);
  track.addEventListener('pointerenter',()=>clearInterval(timer));
  track.addEventListener('pointerleave',()=>timer=setInterval(next,interval));
}

// ==========================
// Mark Flash Badges on other pages
// ==========================
function markFlashBadges(offers){
  if(!offers || !offers.length) return;
  const now=new Date().getTime();

  offers.forEach(o=>{
    const end=new Date(o.deadline).getTime();
    if(end>now){
      const linkEls=document.querySelectorAll(`a[href="${o.link}"], a[href*="${o.link}"]`);
      linkEls.forEach(a=>{
        if(!a.closest('.flash-badge')){
          const badge=document.createElement('span');
          badge.className='flash-badge';
          badge.textContent='⚡ Flash Offer';
          a.insertAdjacentElement('beforebegin',badge);
        }
      });
    }
  });
}

✅ How it works

When your website loads:

flash-offer.js fetches flash-offers.json

Creates Flash cards automatically inside your slider

Adds countdown timers

Checks all links on the page; if they match a current offer, it adds a “⚡ Flash Offer” badge next to them

When an offer’s date expires:

It disappears from the Flash Offers section

Badges are automatically removed site-wide

You can update offers anytime by editing only the JSON file — super easy.
