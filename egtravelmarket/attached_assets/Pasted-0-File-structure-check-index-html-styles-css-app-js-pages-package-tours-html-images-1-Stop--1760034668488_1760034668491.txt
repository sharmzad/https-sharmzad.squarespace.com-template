D) Bullet-proof fixer script (handles any leftover “/something” links)

Add this just before </body> in both index.html and pages/package-tours.html:

<script>
/* Replit 404 guard: rewrite leading-slash links to relative */
(function () {
  const inSub = location.pathname.includes('/pages/');
  document.querySelectorAll('a[href^="/"]').forEach(a=>{
    const raw = a.getAttribute('href');        // e.g. "/pages/package-tours.html" or "/"
    if (raw === '/') { a.setAttribute('href', inSub ? '../index.html' : './'); return; }
    if (raw.startsWith('/pages/')) { a.setAttribute('href', inSub ? '..'+raw : '.'+raw); return; }
    a.setAttribute('href', (inSub ? '../' : './') + raw.replace(/^\//, ''));
  });
})();
</script>


This guarantees no more 404s from logo/home/deals/menu icons or footer.
