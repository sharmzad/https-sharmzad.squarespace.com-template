/* ==========================

const WEEKLY = {
  tripName: "Cairo by Flight",
  day: "Thursday",
  bonuses: ["Free Turkish Bath", "Local SIM Card"],
  summary: "This week only: Free Turkish Bath + Local SIM Card.",
  heroImg: "https://images.unsplash.com/photo-1544989164-31dc3c645987?q=80&w=1600&auto=format&fit=crop",
  description: "Fly from Sharm El-Sheikh to Cairo for a day of wonder—see the Pyramids of Giza, the Grand Egyptian Museum area, and enjoy a guided experience. This week, we’re boosting value with a relaxing Turkish Bath and a free local SIM so you’re always connected.",
  enableWhatsApp: false, // keep false for now; set true later to open WhatsApp on submit
  whatsappNumber: "201023889959" // digits only, no + (your WhatsApp)
};

/* ====== INIT CONTENT ====== */
function initWeeklyContent(){
  // Hero & meta
  const $ = (sel)=>document.querySelector(sel);
  $("#tripTitle").textContent = WEEKLY.tripName;
  $("#tripSummary").textContent = WEEKLY.summary;
  $("#heroImg").src = WEEKLY.heroImg;
  $("#metaTripName").textContent = WEEKLY.tripName;
  $("#metaTripDay").textContent = WEEKLY.day;
  $("#tripDay").value = WEEKLY.day;
  $("#metaBonuses").textContent = WEEKLY.bonuses.join(", ");
  $("#bonuses").value = WEEKLY.bonuses.join(", ");
  $("#tripDescription").textContent = WEEKLY.description;
  $("#tripName").value = WEEKLY.tripName;

  // Badges
  const bb = $("#bonusBadges");
  bb.innerHTML = "";
  WEEKLY.bonuses.forEach(b=>{
    const s = document.createElement("span");
    s.className = "badge"; s.textContent = b;
    bb.appendChild(s);
  });

  // Price labels (the inputs below are editable by you each week)
  const pa = document.getElementById("priceAdult");
  const pc = document.getElementById("priceChild");
  document.getElementById("priceAdultLabel").textContent = `$${Number(pa.value||0)}`;
  document.getElementById("priceChildLabel").textContent = `$${Number(pc.value||0)}`;

  // Date - set min today
  const today = new Date();
  const y = today.getFullYear();
  const m = String(today.getMonth()+1).padStart(2,"0");
  const d = String(today.getDate()).padStart(2,"0");
  document.getElementById("tripDate").min = `${y}-${m}-${d}`;

  // Footer year
  document.getElementById("year").textContent = y;

  updateTotal();
}

/* ====== TOTAL CALC ====== */
function updateTotal(){
  const A = Number(document.getElementById("adults").value || 0);
  const C = Number(document.getElementById("children").value || 0);
  const pA = Number(document.getElementById("priceAdult").value || 0);
  const pC = Number(document.getElementById("priceChild").value || 0);
  const total = (A * pA) + (C * pC);

  document.getElementById("calcLabel").textContent = `$${total}`;
  document.getElementById("totalLabel").textContent = `Total: $${total}`;
}

/* ====== FORM HANDLERS ====== */
["adults","children","priceAdult","priceChild"].forEach(id=>{
  document.addEventListener("change", e=>{
    if(e.target && e.target.id === id){
      // reflect price in hero chips if price fields changed
      if(id === "priceAdult"){
        document.getElementById("priceAdultLabel").textContent = `$${Number(e.target.value||0)}`;
      }
      if(id === "priceChild"){
        document.getElementById("priceChildLabel").textContent = `$${Number(e.target.value||0)}`;
      }
      updateTotal();
    }
  }, true);
});

document.getElementById("towForm").addEventListener("submit", function(e){
  e.preventDefault();

  // Collect data (ready for backend later)
  const fd = new FormData(e.target);
  const payload = Object.fromEntries(fd.entries());

  // Option A: Confirmation popup (as agreed)
  document.getElementById("okPopup").classList.add("open");

  // Optional (OFF): WhatsApp redirect with prefilled message
  if(WEEKLY.enableWhatsApp){
    const msg =
`Trip of the Week Booking:
Trip: ${payload.tripName}
Date: ${payload.tripDate} (${payload.tripDay})
Bonuses: ${payload.bonuses}
Adults: ${payload.adults} @ $${payload.priceAdult}
Children: ${payload.children} @ $${payload.priceChild}
Name: ${payload.name}
Phone: ${payload.phone}
Email: ${payload.email}
Notes: ${payload.notes || "-"}`;
    const url = `https://wa.me/${WEEKLY.whatsappNumber}?text=${encodeURIComponent(msg)}`;
    window.open(url, "_blank");
  }

  // For now we keep it client-side. Later: POST to your backend/email.
});

document.getElementById("printSummary").addEventListener("click", ()=>{
  window.print();
});

document.getElementById("okPopup").addEventListener("click", (e)=>{
  // close when clicking background
  if(e.target.id === "okPopup"){ e.currentTarget.classList.remove("open"); }
});

initWeeklyContent();

How to use / weekly updates

Change the weekly offer by editing the small WEEKLY config at the top of the <script>:

tripName, day, bonuses, summary, heroImg, description

Set the weekly price inside the form (as you asked):

In the Price (Adult) and Price (Child) fields at the top of the form—just type the values for this week.

Total updates automatically based on Adults/Children selected.

Confirmation popup is active (Option A ✅).

To enable WhatsApp handoff later, set enableWhatsApp: true.
