<script>
// Get token and email from URL
const params = new URLSearchParams(window.location.search);
const token = params.get('token');
const email = params.get('email');

const verifyingEl = document.getElementById('verifying');
const successEl = document.getElementById('success');
const errorEl = document.getElementById('error');
const errorMessageEl = document.getElementById('error-message');

if (!token || !email) {
  errorMessageEl.textContent = 'Error: Invalid verification link. No token or email provided.';
  errorEl.style.display = 'block';
} else {
  verifyingEl.style.display = 'block';

  fetch('/api/auth/verify-email', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ 
      email: decodeURIComponent(email),
      token 
    })
  })
  .then(res => res.json())
  .then(data => {
    verifyingEl.style.display = 'none';
    
    if (data.message && data.message.includes('successfully')) {
      successEl.style.display = 'block';
    } else {
      errorMessageEl.textContent = data.message || 'Failed to verify email';
      errorEl.style.display = 'block';
    }
  })
  .catch(err => {
    verifyingEl.style.display = 'none';
    errorMessageEl.textContent = 'Error: ' + err.message;
    errorEl.style.display = 'block';
  });
}
</script>
