// Preselect package from URL (?pkg=...)
(function(){
  const params = new URLSearchParams(location.search);
  const fromPage = params.get('pkg');
  if(fromPage){
    const select = document.getElementById('package');
    [...select.options].forEach(opt=>{
      if(opt.text.trim() === decodeURIComponent(fromPage).trim()){
        select.value = opt.text; 
      }
    });
  }
})();

function handleBooking(e){
  e.preventDefault();
  const requiredIds = ['package','startDate','adults','fullname','email','mobile','pickup'];
  for(const id of requiredIds){
    const el = document.getElementById(id);
    if(!el.value || (el.type==='number' && +el.value<=0)){
      document.getElementById('bookingMessage').textContent = 'Please fill all required fields correctly.';
      el.focus();
      return false;
    }
  }
  const payload = {
    package: document.getElementById('package').value,
    startDate: document.getElementById('startDate').value,
    accessCode: document.getElementById('accessCode').value,
    adults: document.getElementById('adults').value,
    children: document.getElementById('children').value,
    mobile: document.getElementById('mobile').value,
    fullname: document.getElementById('fullname').value,
    email: document.getElementById('email').value,
    pickup: document.getElementById('pickup').value,
    notes: document.getElementById('notes').value,
