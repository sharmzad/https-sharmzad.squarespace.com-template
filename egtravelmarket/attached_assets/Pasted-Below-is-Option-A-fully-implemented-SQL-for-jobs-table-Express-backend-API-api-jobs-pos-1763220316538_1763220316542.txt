1️⃣ Database – create jobs table

In your Postgres (where you created your other tables), run this SQL once:

CREATE TABLE IF NOT EXISTS jobs (
    id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    job_type VARCHAR(50) NOT NULL, -- 'guide' | 'diver' | 'mentor'
    location VARCHAR(255) NOT NULL,
    salary VARCHAR(255),
    whatsapp VARCHAR(50),
    email VARCHAR(255),
    description TEXT NOT NULL,
    posted_by VARCHAR(255) NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    status VARCHAR(50) DEFAULT 'approved' -- keep it simple: auto-approved
);


If you have a schema.sql file in Replit, add this there too.

2.1 Create routes/jobs.js

In Replit, create a new file: routes/jobs.js and paste:

// routes/jobs.js
const express = require('express');
const router = express.Router();
const db = require('../db'); // adjust the path if your db file is somewhere else

// POST /api/jobs  -> create a new job (public)
router.post('/', async (req, res) => {
  try {
    const {
      title,
      jobType,
      location,
      salary,
      whatsapp,
      email,
      description,
      postedBy,
    } = req.body;

    // Basic validation
    if (!title || !jobType || !location || !description || !postedBy) {
      return res.status(400).json({ error: 'Missing required fields.' });
    }

    const result = await db.query(
      `INSERT INTO jobs
        (title, job_type, location, salary, whatsapp, email, description, posted_by)
       VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
       RETURNING *`,
      [title, jobType, location, salary || '', whatsapp || '', email || '', description, postedBy]
    );

    res.status(201).json({ success: true, job: result.rows[0] });
  } catch (err) {
    console.error('Error inserting job:', err);
    res.status(500).json({ error: 'Server error while creating job.' });
  }
});

// GET /api/jobs  -> list approved jobs with optional filters
router.get('/', async (req, res) => {
  try {
    const { type, location } = req.query;

    let sql = `SELECT id, title, job_type, location, salary, whatsapp, email, description, posted_by, created_at
               FROM jobs
               WHERE status = 'approved'`;
    const params = [];

    if (type) {
      params.push(type);
      sql += ` AND job_type = $${params.length}`;
    }

    if (location) {
      params.push(`%${location}%`);
      sql += ` AND location ILIKE $${params.length}`;
    }

    sql += ' ORDER BY created_at DESC';

    const result = await db.query(sql, params);
    res.json({ jobs: result.rows });
  } catch (err) {
    console.error('Error fetching jobs:', err);
    res.status(500).json({ error: 'Server error while fetching jobs.' });
  }
});

module.exports = router;

2.2 Connect router in your main server file

In server.js or app.js (where you create your Express app), add:

const jobsRouter = require('./routes/jobs');
app.use('/api/jobs', jobsRouter);


Make sure this is after app.use(express.json()).

  <script>
    const jobsContainer = document.getElementById('jobsContainer');
    const emptyState = document.getElementById('emptyState');
    const filterType = document.getElementById('filterType');
    const filterLocation = document.getElementById('filterLocation');
    const filterBtn = document.getElementById('filterBtn');
    const clearBtn = document.getElementById('clearBtn');

    async function fetchJobs() {
      const params = new URLSearchParams();
      if (filterType.value) params.append('type', filterType.value);
      if (filterLocation.value.trim()) params.append('location', filterLocation.value.trim());

      const url = '/api/jobs' + (params.toString() ? `?${params.toString()}` : '');

      jobsContainer.innerHTML = '';
      emptyState.style.display = 'none';

      try {
        const res = await fetch(url);
        const data = await res.json();

        if (!res.ok) {
          throw new Error(data.error || 'Failed to load jobs.');
        }

        renderJobs(data.jobs || []);
      } catch (err) {
        console.error(err);
        emptyState.textContent = 'Error loading jobs. Please try again later.';
        emptyState.style.display = 'block';
      }
    }

    function renderJobs(jobs) {
      jobsContainer.innerHTML = '';

      if (!jobs.length) {
        emptyState.textContent = 'No jobs found. Try changing the filters.';
        emptyState.style.display = 'block';
        return;
      }

      emptyState.style.display = 'none';

      jobs.forEach(job => {
        const card = document.createElement('article');
        card.className = 'job-card';

        const badge = document.createElement('div');
        badge.className = `job-badge ${job.job_type}`;
        badge.textContent = job.job_type.toUpperCase();
        card.appendChild(badge);

        const titleEl = document.createElement('h2');
        titleEl.className = 'job-title';
        titleEl.textContent = job.title;
        card.appendChild(titleEl);

        const metaEl = document.createElement('div');
        metaEl.className = 'job-meta';
        metaEl.textContent = `${job.location} • Posted by ${job.posted_by}`;
        card.appendChild(metaEl);

        if (job.salary) {
          const salaryEl = document.createElement('div');
          salaryEl.className = 'job-salary';
          salaryEl.textContent = job.salary;
          card.appendChild(salaryEl);
        }

        const descEl = document.createElement('div');
        descEl.className = 'job-desc';
        descEl.textContent = job.description.length > 180
          ? job.description.slice(0, 180) + '...'
          : job.description;
        card.appendChild(descEl);

        const footer = document.createElement('div');
        footer.className = 'job-footer';

        const dateSpan = document.createElement('span');
        const created = new Date(job.created_at);
        dateSpan.textContent = `Posted on ${created.toLocaleDateString()}`;
        footer.appendChild(dateSpan);

        card.appendChild(footer);

        const btnRow = document.createElement('div');
        btnRow.className = 'btn-row';

        if (job.whatsapp) {
          const waBtn = document.createElement('button');
          waBtn.className = 'btn btn-whatsapp';
          waBtn.textContent = 'Apply via WhatsApp';
          waBtn.addEventListener('click', () => {
            openWhatsApp(job);
          });
          btnRow.appendChild(waBtn);
        }

        if (job.email) {
          const emailBtn = document.createElement('button');
          emailBtn.className = 'btn btn-email';
          emailBtn.textContent = 'Apply via Email';
          emailBtn.addEventListener('click', () => {
            openEmail(job);
          });
          btnRow.appendChild(emailBtn);
        }

        card.appendChild(btnRow);
        jobsContainer.appendChild(card);
      });
    }

    function openWhatsApp(job) {
      const raw = job.whatsapp || '';
      const cleaned = raw.replace(/\D/g, ''); // keep only digits
      if (!cleaned) return;

      const text = `Hello, I am interested in the job "${job.title}" in ${job.location} that you posted on EgTravelMarket.`;
      const url = `https://wa.me/${cleaned}?text=${encodeURIComponent(text)}`;
      window.open(url, '_blank');
    }

    function openEmail(job) {
      const subject = `Application for: ${job.title}`;
      const body = `Hello,\n\nI am interested in the job "${job.title}" in ${job.location} that you posted on EgTravelMarket.\n\nBest regards,\n`;
      const url = `mailto:${encodeURIComponent(job.email)}?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
      window.location.href = url;
    }

    filterBtn.addEventListener('click', fetchJobs);
    clearBtn.addEventListener('click', () => {
      filterType.value = '';
      filterLocation.value = '';
      fetchJobs();
    });

    // Load jobs on first visit
    fetchJobs();
