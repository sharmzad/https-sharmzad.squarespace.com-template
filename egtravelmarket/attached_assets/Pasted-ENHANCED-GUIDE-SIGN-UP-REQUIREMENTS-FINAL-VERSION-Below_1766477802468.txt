5️⃣ SAFE JS (SHOW / HIDE “OTHER” & LICENSE NUMBER)
<script>
document.getElementById('languagesSelect').addEventListener('change', function () {
  const selected = Array.from(this.selectedOptions).map(o => o.value);
  document.getElementById('languageOther').style.display =
    selected.includes('Other') ? 'block' : 'none';
});

document.querySelectorAll('input[name="has_tourism_license"]').forEach(radio => {
  radio.addEventListener('change', function () {
    document.getElementById('licenseNumber').style.display =
      this.value === 'yes' ? 'block' : 'none';
  });
});
</script>
