<script>
const API_URL = window.location.origin + '/api';

let authToken = localStorage.getItem('authToken');
let currentUser = null;

if (!authToken) {
  window.location.href = '/agency-login.html';
}

console.log('✅ Script loaded, API_URL:', API_URL);

function formatDate(dateString) {
  const date = new Date(dateString);
  return date.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
}

function formatSalary(min, max) {
  if (!min && !max) return 'Not specified';
  if (min && max) return `$${min} - $${max}`;
  if (min) return `From $${min}`;
  return `Up to $${max}`;
}

async function loadProfile() {
  try {
    const response = await fetch(`${API_URL}/auth/me`, {
      headers: { 'Authorization': `Bearer ${authToken}` }
    });

    if (!response.ok) {
      throw new Error('Failed to load profile');
    }

    const responseData = await response.json();
    currentUser = responseData.user || responseData;  // Handle both response formats
    const profile = responseData.agencyProfile || responseData.agency_profile || currentUser.agencyProfile || currentUser.agency_profile || {};

    // Display profile info
    document.getElementById('profile-company-name').textContent = profile.companyName || profile.company_name || 'N/A';
    document.getElementById('profile-email').textContent = currentUser.email || 'N/A';
    document.getElementById('profile-description').textContent = profile.companyDescription || profile.company_description || 'Not provided';
    document.getElementById('profile-location').textContent = profile.location || 'Not provided';
    
    // Handle arrays/strings
    const services = Array.isArray(profile.services) ? profile.services.join(', ') : profile.services || 'Not provided';
    const areas = Array.isArray(profile.operatingAreas || profile.operating_areas) ? (profile.operatingAreas || profile.operating_areas).join(', ') : (profile.operatingAreas || profile.operating_areas || 'Not provided');
    
    document.getElementById('profile-services').textContent = services;
    document.getElementById('profile-areas').textContent = areas;
    document.getElementById('profile-years').textContent = profile.yearsInBusiness || profile.years_in_business || 'Not provided';
    document.getElementById('profile-license').textContent = profile.licenseNumber || profile.license_number || 'Not provided';
    document.getElementById('profile-website').textContent = profile.websiteUrl || profile.website_url || 'Not provided';
    document.getElementById('profile-whatsapp').textContent = profile.whatsapp || 'Not provided';
    
    const statusBadge = document.getElementById('profile-status');
    statusBadge.textContent = currentUser.approvalStatus || currentUser.approval_status || 'unknown';

async function loadApplications() {
  try {
    const response = await fetch(`${API_URL}/jobs/my-jobs`, {
      headers: { 'Authorization': `Bearer ${authToken}` }
    });

    if (!response.ok) {
      throw new Error('Failed to load jobs');
    }

    const data = await response.json();
    const jobs = data.jobs || data || [];
    const container = document.getElementById('applications-container');

    const jobsWithApplications = jobs.filter(j => j.application_count > 0 || j.totalApplications > 0);

    if (jobsWithApplications.length === 0) {

          <p>No applications received yet.</p>
        </div>
      `;
      return;

    let html = '';
    for (const job of jobsWithApplications) {

        headers: { 'Authorization': `Bearer ${authToken}` }

                  <div class="application-info">
                    <div class="application-name">
                      <a href="/${app.expert?.expertType}/${app.expert?.slug}" target="_blank" style="color: var(--primary); text-decoration: none; font-weight: 600;">
                        ${app.expert?.displayName || app.expert?.fullName || 'Anonymous'} 🔗
                      </a>
                      ${app.expert?.profilePhoto ? `<img loading="lazy" src="${app.expert.profilePhoto}" style="width: 40px; height: 40px; border-radius: 50%; margin-left: 8px; vertical-align: middle;">` : ''}
                    </div>
                    <div class="application-meta">
                      ${app.expert?.expertType ? `<span class="status-badge status-pending">${app.expert.expertType.toUpperCase()}</span> • ` : ''}

                    ${app.expert?.phone ? `<div style="font-size: 13px;">📱 ${app.expert.phone}</div>` : ''}
                    ${app.expert?.whatsapp ? `<div style="font-size: 13px;">💬 WhatsApp: ${app.expert.whatsapp}</div>` : ''}
                  </div>
                  <div class="action-buttons">
                    ${app.status === 'applied' ? `
                      <button class="btn-small approve" onclick="updateApplicationStatus(${app.id}, 'hired')">✅ Hire</button>
                      <button class="btn-small reject" onclick="updateApplicationStatus(${app.id}, 'rejected')">❌ Reject</button>
                      <button class="btn-small" onclick="updateApplicationStatus(${app.id}, 'contacted')">📞 Mark as Contacted</button>

async function closeJob(jobId) {
  if (!confirm('Are you sure you want to close this job posting?')) return;

  try {
    const response = await fetch(`${API_URL}/jobs/${jobId}`, {
      method: 'DELETE',
      headers: { 'Authorization': `Bearer ${authToken}` }
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || 'Failed to close job');
    }

    alert('Job closed successfully');
    loadMyJobs();
  } catch (error) {
    alert('Error: ' + error.message);
  }
}

async function updateApplicationStatus(applicationId, status) {
  try {
    const response = await fetch(`${API_URL}/applications/${applicationId}/status`, {
      method: 'PUT',
      headers: {
        'Authorization': `Bearer ${authToken}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ status })
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || 'Failed to update status');
    }

    alert('Application status updated');
    loadApplications();
  } catch (error) {
    alert('Error: ' + error.message);
  }
}

function showAlert(containerId, message, type = 'success') {
  const container = document.getElementById(containerId);
  container.innerHTML = `<div class="alert alert-${type}">${message}</div>`;
  setTimeout(() => { container.innerHTML = ''; }, 5000);
}

function switchTab(tabName) {
  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.tab-content').forEach(tc => tc.classList.remove('active'));
  
  const tab = document.querySelector(`[data-tab="${tabName}"]`);
  tab.classList.add('active');
  document.getElementById(`${tabName}-tab`).classList.add('active');
  
  if (tabName === 'myjobs') loadMyJobs();
  if (tabName === 'applications') loadApplications();
}

document.querySelectorAll('.tab').forEach(tab => {
  tab.addEventListener('click', () => {
    switchTab(tab.dataset.tab);
  });
});

async function loadPayouts() {

  }
