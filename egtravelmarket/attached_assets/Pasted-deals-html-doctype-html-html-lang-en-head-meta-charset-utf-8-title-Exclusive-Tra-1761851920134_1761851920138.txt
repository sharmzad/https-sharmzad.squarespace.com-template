flash-offer.js
// ===== Countdown for all .countdown elements with [data-deadline] =====
(function () {
  const els = Array.from(document.querySelectorAll('.countdown[data-deadline]'));
  function two(n){ return n < 10 ? '0' + n : '' + n; }

  function tick() {
    const now = new Date().getTime();
    els.forEach(el => {
      const deadlineStr = el.getAttribute('data-deadline');
      const end = new Date(deadlineStr).getTime();
      const diff = end - now;

      if (isNaN(end)) { el.textContent = '—'; return; }

      if (diff <= 0) {
        el.textContent = 'Offer ended';
        el.closest('.flash-card')?.classList.add('is-ended');
        return;
      }

      const d = Math.floor(diff / (1000 * 60 * 60 * 24));
      const h = Math.floor((diff / (1000 * 60 * 60)) % 24);
      const m = Math.floor((diff / (1000 * 60)) % 60);

      el.textContent = `Ends in ${two(d)}d ${two(h)}h ${two(m)}m`;
    });
  }

  tick();
  setInterval(tick, 30 * 1000); // update every 30s (minutes precision is enough)
})();

// ===== Simple auto-scroll for the flash slider (optional) =====
(function () {
  const track = document.querySelector('.flash-track[data-autoplay="true"]');
  if (!track) return;

  const interval = parseInt(track.getAttribute('data-interval') || '6000', 10);
  const cards = Array.from(track.children);
  if (cards.length <= 1) return;

  let index = 0;
  function goNext() {
    index = (index + 1) % cards.length;
    cards[index].scrollIntoView({ behavior: 'smooth', inline: 'start', block: 'nearest' });
  }

  let timer = setInterval(goNext, interval);

  // Pause on user interaction
  track.addEventListener('pointerenter', () => clearInterval(timer));
  track.addEventListener('pointerleave', () => (timer = setInterval(goNext, interval)));
})();

Create the files on Replit:

New file deals.html (paste HTML above)

New file deals.css (paste CSS)

New file flash-offer.js (paste JS)

Reuse your header & footer

In deals.html, replace the placeholders in <header id="site-header"> and <footer id="site-footer"> with your existing header/footer HTML (exactly as used on your homepage).

Link it from your menu icon

In your nav, set the Deals icon link to /deals.html.

Add your widgets

Replace the “widget placeholder” boxes with your Aviasales/WayAway flight script and Booking/Travelpayouts hotel script.

Edit deadlines & links

In the Flash cards, change each data-deadline (ISO format with your timezone, e.g. +02:00) and update the Book Now links to your real pages.
