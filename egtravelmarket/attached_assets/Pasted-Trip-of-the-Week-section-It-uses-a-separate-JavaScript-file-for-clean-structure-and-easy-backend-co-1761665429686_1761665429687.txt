// --- PAGE LOGIC ---
const $ = s => document.querySelector(s);
const storeKey = "tow_seats_left";
let codeOK = false;

document.addEventListener("DOMContentLoaded", () => {
  initTrip();
  $("#checkCode").addEventListener("click", verifyCode);
});

function initTrip() {
  $("#tripTitle").textContent = WEEKLY.tripName;
  $("#tripSummary").textContent = WEEKLY.summary;
  $("#heroImg").src = WEEKLY.heroImg;
  $("#chipDay").textContent = `Main Day: ${WEEKLY.mainDay}`;
  $("#chipType").textContent = WEEKLY.isAirTrip ? "Air Trip" : "Land/Sea";
  $("#year").textContent = new Date().getFullYear();

  // Seats
  let seats = Number(localStorage.getItem(storeKey));
  if (!seats || seats <= 0 || seats > WEEKLY.seats) {
    seats = WEEKLY.seats;
    localStorage.setItem(storeKey, seats);
  }
  updateSeatsUI(seats);

  $("#tripDescription").textContent = WEEKLY.description;

  // itinerary table
  const tb = $("#itineraryBody");
  tb.innerHTML = "";
  WEEKLY.itinerary.forEach(i => {
    const tr = document.createElement("tr");
    tr.innerHTML = `<td>${i.day}</td><td>${i.title}</td><td>${i.notes}</td>`;
    tb.appendChild(tr);
  });

  // includes
  const inc = $("#includes");
  inc.innerHTML = "";
  WEEKLY.includes.forEach(t => {
    const s = document.createElement("span");
    s.className = "pill";
    s.textContent = t;
    inc.appendChild(s);
  });

  // form base
  $("#towForm").innerHTML = `
    <h3>This Week’s Prices</h3>
    <div class="row"><div class="field"><label>Adult</label><div class="price-display">$${WEEKLY.priceAdult}</div></div>
    <div class="field"><label>Child</label><div class="price-display">$${WEEKLY.priceChild}</div></div></div>
    <h3>Passengers</h3>
    <div class="field"><label>Adults</label><input id="adults" type="number" min="0" max="10" value="1" disabled></div>
    <div class="field"><label>Children</label><input id="children" type="number" min="0" max="10" value="0" disabled></div>
    <div class="field"><label>Infants (FREE)</label><input id="infants" type="number" min="0" max="4" value="0" disabled></div>
    <h3>Hotel & Contact</h3>
    <div class="field"><label>Hotel Name</label><input id="hotelName" disabled></div>
    <div class="field"><label>Room Number</label><input id="roomNumber" disabled></div>
    <div class="field"><label>Name</label><input id="name" disabled></div>
    <div class="field"><label>Phone (WhatsApp)</label><input id="phone" disabled></div>
    <div class="field"><label>Email</label><input id="email" type="email" disabled></div>
    <button id="payBtn" class="btn" disabled>Proceed to Payment</button>
    <div id="summaryArea" style="margin-top:10px;"></div>
  `;
  $("#payBtn").addEventListener("click", proceedToPayment);
}

function updateSeatsUI(n) {
  $("#chipSeats").textContent = `Seats Left: ${n}`;
  $("#seatInfo").textContent = n > 0
    ? `Only ${n} seats left this week. Enter access code to book.`
    : "Sold out this week.";
}

function verifyCode() {
  const code = ($("#accessCode").value || "").trim();
  let seats = Number(localStorage.getItem(storeKey)) || WEEKLY.seats;
  if (seats <= 0) { $("#codeFeedback").textContent = "Sold out."; return; }
  if (!code) { $("#codeFeedback").textContent = "Enter your code."; return; }

  if (WEEKLY.accessCodes.includes(code)) {
    codeOK = true;
    $("#codeFeedback").textContent = "✓ Code accepted! Fill your details.";
    ["adults","children","infants","hotelName","roomNumber","name","phone","email"].forEach(id => $("#"+id).disabled=false);
    $("#payBtn").disabled=false;
  } else {
    codeOK = false;
    $("#codeFeedback").textContent = "Invalid code.";
  }
}

function proceedToPayment() {
  if (!codeOK) return alert("Enter a valid access code first.");
  let seats = Number(localStorage.getItem(storeKey));
  const A = +$("#adults").value, C = +$("#children").value;
  const pax = A + C;
  if (pax <= 0) return alert("Select at least one passenger.");
  if (pax > seats) return alert(`Only ${seats} seat(s) left.`);

  seats -= pax; localStorage.setItem(storeKey, seats); updateSeatsUI(seats);

  const params = new URLSearchParams({
    trip: WEEKLY.tripName, adults:A, children:C,
    priceAdult:WEEKLY.priceAdult, priceChild:WEEKLY.priceChild,
    total:(A*WEEKLY.priceAdult)+(C*WEEKLY.priceChild),
    hotelName:$("#hotelName").value, roomNumber:$("#roomNumber").value,
    phone:$("#phone").value, email:$("#email").value, name:$("#name").value
  });
  window.open(`${WEEKLY.paymentUrl}?${params.toString()}`, "_blank");
}
