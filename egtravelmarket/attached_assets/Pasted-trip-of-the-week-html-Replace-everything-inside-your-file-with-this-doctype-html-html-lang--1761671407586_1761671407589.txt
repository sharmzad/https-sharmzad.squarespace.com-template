// --- CONFIG ---
const WEEKLY = {
  tripName: "Cairo by Flight",
  mainDay: "Tuesday",
  priceAdult: 180,
  priceChild: 160,
  itinerary: [
    { day:"Tuesday", title:"Transfer from Hotel to Sharm Airport", notes:"Pick-up 04:00–05:00 AM.", type:"main"},
    { day:"Tuesday", title:"Flight to Cairo", notes:"1-hour flight / Meet guide on arrival.", type:"main"},
    { day:"Tuesday", title:"Pyramids & Sphinx Tour", notes:"Guided visit + photo stops.", type:"main"},
    { day:"Tuesday", title:"Lunch by the Nile", notes:"Buffet meal with river view.", type:"main"},
    { day:"Tuesday", title:"Egyptian Museum / Grand Museum", notes:"Explore ancient treasures.", type:"main"},
    { day:"Tuesday", title:"Return Flight to Sharm", notes:"Evening flight + hotel transfer.", type:"main"},
    { day:"Tuesday", title:"SIM Card 20 GB (Free Bonus)", notes:"Delivered Day 1.", type:"bonus"},
    { day:"Thursday", title:"Turkish Bath 15 min (Free Bonus)", notes:"Relaxation session.", type:"bonus"}
  ],
  includes: [
    "Round-trip Flights (Sharm ↔ Cairo)",
    "Air-conditioned Transfers",
    "Lunch by the Nile",
    "Entrance Fees",
    "Professional Guide",
    "SIM Card 20 GB (Bonus)",
    "15-min Turkish Bath (Bonus)"
  ]
};

// --- UTIL ---
const $ = s => document.querySelector(s);

// --- LOAD DATA ON START ---
document.addEventListener("DOMContentLoaded", () => {
  fillItinerary();
  fillIncludes();
  loadAvailableDates();
  $("#year").textContent = new Date().getFullYear();
  $("#payBtn").addEventListener("click", proceedToPayment);
});

// itinerary table
function fillItinerary() {
  const tb = $("#itineraryBody");
  tb.innerHTML = "";
  WEEKLY.itinerary.forEach(i => {
    const tr = document.createElement("tr");
    tr.innerHTML = `<td>${i.day}</td><td>${i.title}</td><td>${i.notes}</td>`;
    tb.appendChild(tr);
  });
}

// includes pills
function fillIncludes() {
  const inc = $("#includes");
  WEEKLY.includes.forEach(t => {
    const s = document.createElement("span");
    s.className = "pill";
    s.textContent = t;
    inc.appendChild(s);
  });
}

// --- LOAD AVAILABLE DATES FROM JSON ---
async function loadAvailableDates() {
  try {
    const res = await fetch("/data/trip-availability.json");
    const data = await res.json();
    const dates = data.trip_of_week;
    const sel = $("#tripDate");
    sel.innerHTML = "";
    dates.forEach(d => {
      if (d.seats_left > 0) {
        const dt = new Date(d.date);
        const opt = document.createElement("option");
        opt.value = d.date;
        opt.textContent =
          `${dt.toLocaleDateString("en-GB", {weekday:"long",day:"numeric",month:"short",year:"numeric"})} — ${d.seats_left} seats left`;
        sel.appendChild(opt);
      }
    });
    if (!sel.options.length) {
      const o = document.createElement("option");
      o.textContent = "No available trips this week";
      o.disabled = true;
      sel.appendChild(o);
    }
  } catch (err) {
    console.error("Error loading dates", err);
  }
}

// --- PAYMENT / SUMMARY ---
function proceedToPayment() {
  const name = $("#name").value.trim();
  const phone = $("#phone").value.trim();
  const date = $("#tripDate").value;
  const A = +$("#adults").value;
  const C = +$("#children").value;
  if (!name || !phone) return alert("Please fill your name and WhatsApp number.");
  if (!date) return alert("Please select a trip date.");

  const total = A * WEEKLY.priceAdult + C * WEEKLY.priceChild;
  $("#summaryArea").innerHTML = `
    <div style="margin-top:10px;padding:10px;background:#f1f5ff;border-radius:8px;">
      <strong>Summary</strong><br>
      Trip: ${WEEKLY.tripName}<br>
      Date: ${new Date(date).toLocaleDateString()}<br>
      Adults: ${A} × $${WEEKLY.priceAdult}<br>
      Children: ${C} × $${WEEKLY.priceChild}<br>
      <strong>Total: $${total}</strong><br><br>
      (Payment gateway integration coming soon)
    </div>`;
}
