<script>
/* ==========================
   WEEKLY CONFIG (EDIT HERE)
   ========================== */
const WEEKLY = {
  tripName: "Cairo by Flight",
  mainDay: "Tuesday",
  isAirTrip: true,
  seats: 20, // 🔧 set seats for this week
  accessCodes: ["ETM2025","VIP001","TRIPWEEK"], // 🔧 add/remove codes
  paymentUrl: "https://yourpaymentgateway.com/pay-trip-week", // 🔧 replace with your gateway URL

  summary: "This week only: Free SIM (20GB) + Turkish Bath 15 min.",
  heroImg: "https://images.unsplash.com/photo-1544989164-31dc3c645987?q=80&w=1600&auto=format&fit=crop",
  description:
    "Fly from Sharm El-Sheikh to Cairo for a full-day guided tour: Pyramids of Giza and Grand Egyptian Museum area. Transportation included for all activities. SIM (20GB) is handed to guests on the first activity. Turkish Bath (15 minutes) is planned later for relaxation.",

  // Itinerary: define each activity with its day
  itinerary: [
    { day: "Tuesday", title: "Cairo by Flight", notes: "Main trip with guide, museum area & Giza Plateau", type:"main" },
    { day: "Tuesday", title: "SIM Card 20GB (Free Bonus)", notes: "Delivered on first activity (Day 1)", type:"bonus" },
    { day: "Thursday", title: "Turkish Bath / Massage (15 min, Free Bonus)", notes: "Relaxation session", type:"bonus" }
  ],

  // Inclusions (auto add Transportation; auto add Flight tickets if isAirTrip)
  includes: ["Transportation", "Professional Guide"]
};

/* ===== UTIL ===== */
const $ = (sel)=>document.querySelector(sel);
const storeKey = "tow_seats_left";

/* ===== INIT ===== */
function init(){
  // Header/hero/meta
  $("#tripTitle").textContent = WEEKLY.tripName;
  $("#tripSummary").textContent = WEEKLY.summary;
  $("#heroImg").src = WEEKLY.heroImg;
  $("#metaTripName").textContent = WEEKLY.tripName;
  $("#metaTripDay").textContent = WEEKLY.mainDay;
  $("#chipDay").textContent = `Main Day: ${WEEKLY.mainDay}`;
  $("#tripName").value = WEEKLY.tripName;
  $("#year").textContent = new Date().getFullYear();

  // Seats handling with simple persistence
  let seatsLeft = Number(localStorage.getItem(storeKey));
  if(!seatsLeft || seatsLeft <= 0 || seatsLeft > WEEKLY.seats){
    seatsLeft = WEEKLY.seats;
    localStorage.setItem(storeKey, seatsLeft);
  }
  updateSeatsUI(seatsLeft);

  // Visa note for air trips
  if(WEEKLY.isAirTrip){
    $("#visaNote").style.display = "block";
    $("#chipType").textContent = "Air Trip";
  } else {
    $("#chipType").textContent = "Land/Sea";
  }

  // Bonuses meta line
  const bonusesLine = WEEKLY.itinerary
    .filter(i=>i.type==="bonus")
    .map(i=>`${i.title.replace(" (Free Bonus)","")} (${i.day})`)
    .join(", ");
  $("#metaBonuses").textContent = bonusesLine || "—";

  // Itinerary table
  const tb = $("#itineraryBody");
  tb.innerHTML = "";
  WEEKLY.itinerary.forEach(item=>{
    const tr = document.createElement("tr");
    tr.innerHTML = `
      <td><strong>${item.day}</strong></td>
      <td>${item.title}${item.type==="main" ? " <span class='badge' style='margin-left:6px'>Main</span>" : ""}</td>
      <td class="note">${item.notes || ""}</td>
    `;
    tb.appendChild(tr);
  });

  // Inclusions
  const incWrap = $("#includes");
  incWrap.innerHTML = "";
  const inc = new Set(WEEKLY.includes);
  inc.add("Transportation");
  if(WEEKLY.isAirTrip){ inc.add("Flight tickets"); }
  inc.forEach(txt=>{
    const s = document.createElement("span");
    s.className = "pill";
    s.textContent = txt;
    incWrap.appendChild(s);
  });

  // Date min=today
  const today = new Date();
  const min = `${today.getFullYear()}-${String(today.getMonth()+1).padStart(2,"0")}-${String(today.getDate()).padStart(2,"0")}`;
  $("#tripDate").min = min;

  // listeners
  ["adults","children","priceAdult","priceChild"].forEach(id=>{
    document.getElementById(id).addEventListener("change", updateTotal);
  });
  $("#checkCode").addEventListener("click", verifyCode);
  $("#payBtn").addEventListener("click", proceedToPayment);
  $("#printSummary").addEventListener("click", ()=>window.print());

  updateTotal();
}
function updateSeatsUI(seatsLeft){
  $("#chipSeats").textContent = `Seats Left: ${seatsLeft}`;
  $("#seatInfo").textContent = seatsLeft > 0 ? `Only ${seatsLeft} seats left this week. Enter your access code to unlock payment.` : "Sold out this week.";
  $("#payBtn").disabled = true; // always locked until code ok
}

/* ===== TOTAL ===== */
function updateTotal(){
  const A = Number($("#adults").value||0);
  const C = Number($("#children").value||0);
  const pA = Number($("#priceAdult").value||0);
  const pC = Number($("#priceChild").value||0);
  const total = (A*pA) + (C*pC);
  $("#calcLabel").textContent = `$${total}`;
  $("#totalLabel").textContent = `Total: $${total}`;
}

/* ===== ACCESS CODE ===== */
let codeOK = false;
function verifyCode(){
  const code = ($("#accessCode").value||"").trim();
  const seatsLeft = Number(localStorage.getItem(storeKey)) || WEEKLY.seats;

  if(seatsLeft <= 0){
    $("#codeFeedback").innerHTML = "<span class='err'>Sold out — no seats left this week.</span>";
    $("#payBtn").disabled = true;
    codeOK = false;
    return;
  }
  if(!code){
    $("#codeFeedback").innerHTML = "<span class='err'>Please enter your access code.</span>";
    codeOK = false; $("#payBtn").disabled = true; return;
  }
  if(WEEKLY.accessCodes.includes(code)){
    $("#codeFeedback").innerHTML = "<span class='ok'>Code accepted. You can proceed to payment.</span>";
    codeOK = true;
    // Additional sanity: ensure passengers <= seats
    const pax = Number($("#adults").value||0) + Number($("#children").value||0);
    $("#payBtn").disabled = pax <= 0 ? true : false;
  }else{
    $("#codeFeedback").innerHTML = "<span class='err'>Invalid code. Please check and try again.</span>";
    codeOK = false; $("#payBtn").disabled = true;
  }
}

/* ===== PAYMENT ===== */
function proceedToPayment(){
  // Validate essentials
  if(!codeOK){ $("#codeFeedback").innerHTML = "<span class='err'>Enter a valid access code first.</span>"; return; }
  const req = ["tripDate","priceAdult","priceChild","phone","email","name"];
  for(const id of req){
    const el = document.getElementById(id);
    if(!el.value){ el.focus(); return; }
  }
  // Seats check
  let seatsLeft = Number(localStorage.getItem(storeKey)) || WEEKLY.seats;
  const pax = Number($("#adults").value||0) + Number($("#children").value||0);
  if(pax <= 0){ alert("Please select at least 1 passenger."); return; }
  if(pax > seatsLeft){
    $("#codeFeedback").innerHTML = `<span class='err'>Only ${seatsLeft} seat(s) left. Reduce passengers or try next week.</span>`;
    return;
  }

  // Deduct seats locally (non-authoritative, but prevents double tries on this device)
  seatsLeft = seatsLeft - pax;
  localStorage.setItem(storeKey, seatsLeft);
  updateSeatsUI(seatsLeft);

  // Build a lightweight query for the gateway (optional)
  const params = new URLSearchParams({
    trip: WEEKLY.tripName,
    date: $("#tripDate").value,
    adults: $("#adults").value,
    children: $("#children").value,
    priceAdult: $("#priceAdult").value,
    priceChild: $("#priceChild").value,
    phone: $("#phone").value,
    email: $("#email").value,
    name: $("#name").value
  });

  // Open payment page
  const payURL = WEEKLY.paymentUrl || "#";
  window.open(`${payURL}?${params.toString()}`, "_blank");

  // Show guidance popup
  $("#okPopup").classList.add("open");
}

/* ===== START ===== */
init();
</script>

Open trip-of-the-week.html and update the WEEKLY config at the top of the <script>:

tripName, mainDay, isAirTrip, seats, accessCodes, paymentUrl

heroImg, summary, description

itinerary (set the days for the main trip and bonuses)

Set Adult/Child prices at the top of the form (manual, as you requested).
