<script>
const BACKEND_URL = 'http://' + window.location.hostname + ':3000/api';
let token = localStorage.getItem('authToken');
let allTrips = [];
let filteredStatus = 'all';

async function loadTrips() {
  try {
    const response = await fetch(`${BACKEND_URL}/expert-trips/my-trips`, {
      headers: { 'Authorization': `Bearer ${token}` }
    });

    if (!response.ok) throw new Error('Failed to load trips');
    const data = await response.json();
    allTrips = data.trips || [];
    updateStats();
    displayTrips(allTrips);
  } catch (error) {
    console.error('Error loading trips:', error);
    document.getElementById('loadingSpinner').style.display = 'none';
    document.getElementById('emptyState').style.display = 'block';
  }
}

function updateStats() {
  const draftCount = allTrips.filter(t => t.status === 'draft').length;
  const publishedCount = allTrips.filter(t => t.status === 'published').length;
  const totalBookings = allTrips.reduce((sum, t) => sum + (t.bookings || 0), 0);

  document.getElementById('draftCount').textContent = draftCount;
  document.getElementById('publishedCount').textContent = publishedCount;
  document.getElementById('bookingCount').textContent = totalBookings;
  document.getElementById('earningsAmount').textContent = '$' + (allTrips.reduce((sum, t) => sum + parseFloat(t.price || 0) * 0.8, 0)).toFixed(2);
}

function displayTrips(trips) {
  const spinner = document.getElementById('loadingSpinner');
  const container = document.getElementById('tripsContainer');
  const empty = document.getElementById('emptyState');
  const tbody = document.getElementById('tripsTableBody');

  spinner.style.display = 'none';

  if (trips.length === 0) {
    container.style.display = 'none';
    empty.style.display = 'block';
    return;
  }

  container.style.display = 'block';
  empty.style.display = 'none';

  tbody.innerHTML = trips.map(trip => `
    <tr>
      <td><strong>${trip.title}</strong></td>
      <td>$${parseFloat(trip.price).toFixed(2)}</td>
      <td><span class="status-badge status-${trip.status}">${trip.status.replace(/_/g, ' ').toUpperCase()}</span></td>
      <td>${trip.bookings || 0}</td>
      <td>
        <div class="actions">
          <button class="btn btn-secondary" onclick="viewTripDetails(${trip.id})">View</button>
          ${trip.status === 'draft' ? `<button class="btn btn-primary" onclick="editTrip(${trip.id})">Edit</button>` : ''}
          ${trip.status === 'draft' ? `<button class="btn btn-warning" onclick="submitTrip(${trip.id})">Submit</button>` : ''}
          ${trip.status === 'approved' ? `<button class="btn btn-success" onclick="publishTrip(${trip.id})">Publish</button>` : ''}
        </div>
      </td>
    </tr>
  `).join('');
}

function filterByStatus(status) {
  filteredStatus = status;
  const filtered = status === 'all' ? allTrips : allTrips.filter(t => t.status === status);
  displayTrips(filtered);

  document.querySelectorAll('.tab').forEach(tab => tab.classList.remove('active'));
  event.target.classList.add('active');
}

function openCreateTrip() {
  document.getElementById('createTripModal').classList.add('active');
}

function closeCreateTrip() {
  document.getElementById('createTripModal').classList.remove('active');
  document.querySelector('#createTripModal form').reset();
}

async function handleCreateTrip(e) {
  e.preventDefault();

  const tripData = {
    title: document.getElementById('tripTitle').value,
    description: document.getElementById('tripDescription').value,
    tripType: document.getElementById('tripType').value,
    price: parseFloat(document.getElementById('tripPrice').value),
    location: document.getElementById('tripLocation').value,
    duration: document.getElementById('tripDuration').value,
    maxGroupSize: parseInt(document.getElementById('tripMaxGroup').value) || null,
    startDate: document.getElementById('tripStartDate').value,
    endDate: document.getElementById('tripEndDate').value,
    image: document.getElementById('tripImage').value,
    includedServices: document.getElementById('tripIncluded').value.split(',').map(s => s.trim()).filter(Boolean),
    requirements: document.getElementById('tripRequirements').value.split(',').map(s => s.trim()).filter(Boolean),
  };

  try {
    const response = await fetch(`${BACKEND_URL}/expert-trips/create`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(tripData)
    });

    if (!response.ok) throw new Error('Failed to create trip');
    alert('✅ Trip created successfully as draft!');
    closeCreateTrip();
    loadTrips();
  } catch (error) {
    console.error('Error creating trip:', error);
    alert('Failed to create trip');
  }
}

async function submitTrip(tripId) {
  if (!confirm('Submit this trip for admin approval?')) return;

  try {
    const response = await fetch(`${BACKEND_URL}/expert-trips/${tripId}/submit-approval`, {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${token}` }
    });

    if (!response.ok) throw new Error('Failed to submit');
    alert('✅ Trip submitted for admin approval!');
    loadTrips();
  } catch (error) {
    console.error('Error:', error);
    alert('Failed to submit trip');
  }
}

async function publishTrip(tripId) {
  if (!confirm('Publish this trip to the marketplace?')) return;

  try {
    const response = await fetch(`${BACKEND_URL}/expert-trips/${tripId}/publish`, {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${token}` }
    });

    if (!response.ok) throw new Error('Failed to publish');
    alert('✅ Trip is now live on the marketplace!');
    loadTrips();
  } catch (error) {
    console.error('Error:', error);
    alert('Failed to publish trip');
  }
}

async function viewTripDetails(tripId) {
  const trip = allTrips.find(t => t.id === tripId);
  if (!trip) return;

  const content = document.getElementById('tripDetailsContent');
  content.innerHTML = `
    <p><strong>Title:</strong> ${trip.title}</p>

// Bank Details Form Handler
document.getElementById('bank-details-form').addEventListener('submit', async (e) => {
  e.preventDefault();
  const token = localStorage.getItem('authToken');
  try {
    const response = await fetch('/api/expert-payouts/bank-details', {
      method: 'PUT',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        bankAccountHolder: document.getElementById('bankAccountHolder').value,
        bankName: document.getElementById('bankName').value,
        bankAccountNumber: document.getElementById('bankAccountNumber').value,
        bankIban: document.getElementById('bankIban').value,
        bankCountry: document.getElementById('bankCountry').value
      })
    });
    if (!response.ok) throw new Error('Failed to save bank details');
    document.getElementById('bank-details-status').style.display = 'block';
    setTimeout(() => { document.getElementById('bank-details-status').style.display = 'none'; }, 3000);
  } catch (error) {
    alert('Error: ' + error.message);
  }
});

// Load Bank Details
async function loadBankDetails() {
  const token = localStorage.getItem('authToken');
  try {
    const response = await fetch('/api/expert-payouts/bank-details', {
      headers: { 'Authorization': `Bearer ${token}` }
    });
    if (response.ok) {
      const data = await response.json();
      if (data.bankAccountHolder) {
        document.getElementById('bankAccountHolder').value = data.bankAccountHolder;
        document.getElementById('bankName').value = data.bankName || '';
        document.getElementById('bankCountry').value = data.bankCountry || '';
        document.getElementById('bankAccountNumber').value = '****' + (data.bankAccountNumber ? data.bankAccountNumber.slice(-4) : '');
        document.getElementById('bankIban').value = data.bankIban || '';
      }
    }
  } catch (error) {
    console.error('Error loading bank details:', error);
  }
}

// Load Payouts
async function loadPayouts() {
  const token = localStorage.getItem('authToken');
  try {
    const response = await fetch('/api/expert-payouts/my-payouts', {
      headers: { 'Authorization': `Bearer ${token}` }
    });
    if (response.ok) {
      const data = await response.json();
      const stats = data.stats || {};
      document.getElementById('total-paid').textContent = '$' + (stats.total_paid || 0).toFixed(2);
      document.getElementById('total-pending').textContent = '$' + (stats.total_pending || 0).toFixed(2);
      document.getElementById('total-approved').textContent = '$' + (stats.total_approved || 0).toFixed(2);
      const tbody = document.getElementById('payouts-tbody');
      const payouts = data.payouts || [];
      if (payouts.length === 0) {
        tbody.innerHTML = '<tr><td colspan="5" style="text-align: center; padding: 20px;">No payouts yet!</td></tr>';
        return;
      }
      tbody.innerHTML = payouts.map(p => `
        <tr>
          <td>${p.trip_title || 'Trip'}</td>
          <td>${p.traveler_name || 'N/A'}</td>
          <td>$${parseFloat(p.amount).toFixed(2)}</td>
          <td><span class="status-badge status-${p.status}">${p.status.toUpperCase()}</span></td>
          <td>${new Date(p.created_at).toLocaleDateString()}</td>
        </tr>
      `).join('');
    }
  } catch (error) {
    console.error('Error loading payouts:', error);
  }
}
